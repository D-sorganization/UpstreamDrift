def save(fig,stem,*,dpi):
    fig.canvas.draw(); renderer=fig.canvas.get_renderer(); issues=[]; gaps=[]; cross=[]
    estimate=current['estimate']; marg={n:np.sum(v.density_per_m2,axis=0)*np.diff(estimate.x_edges_m)[0] for n,v in estimate.patterns.items()}
    limit=max(float(np.max(v)) for v in marg.values())*1.05
    for panel,name in enumerate(current['points']):
        main,right=fig.axes[panel*3+1:panel*3+3]
        np.testing.assert_array_equal(right.patches[0].get_data().values,marg[name]);np.testing.assert_array_equal(right.get_xlim(),[0,limit])
        ticks=[t for t in right.get_xticklabels() if t.get_visible() and t.get_text()]
        ticks.sort(key=lambda t:t.get_window_extent(renderer).x0)
        for left,next_ in zip(ticks,ticks[1:]):
            gap=(next_.get_window_extent(renderer).x0-left.get_window_extent(renderer).x1)*72/fig.dpi
            gaps.append({'panel':name,'labels':[left.get_text(),next_.get_text()],'gap_pt':gap})
            if gap<4-1e-8:issues.append('Neighboring marginal tick gutter below 4 pt')
        for a in main.get_xticklabels():
            for b in ticks:
                if a.get_visible() and a.get_text() and a.get_window_extent(renderer).overlaps(b.get_window_extent(renderer)):
                    issues.append('Main and marginal ticks overlap')
        cross.append({'panel':name,'main_tick_labels':[t.get_text() for t in main.get_xticklabels()],'marginal_tick_labels':[t.get_text() for t in ticks]})
    for text in fig.texts:
        box=text.get_window_extent(renderer)
        if box.x0<0 or box.y0<0 or box.x1>fig.bbox.width or box.y1>fig.bbox.height:issues.append('Figure caption outside canvas')
    rows.append({'cell':current['cell'],'presentation':current['mode'],'gaps':gaps,'tick_labels':cross,'issues':issues,'supported_counts':{k:len(v) for k,v in current['points'].items()},'total_counts':current['total'],'density_data_and_limits_exact':True})
    result=save_original(fig,stem,dpi=dpi)
    for path in result.values():artifacts[str(path.relative_to(OUT))]=sha(path)
    png=result['png']
    with Image.open(png) as image:
        preview=OUT/(png.stem+'_1080px.png');image.resize((1080,round(image.height*1080/image.width)),Image.Resampling.LANCZOS).save(preview);previews[str(preview.relative_to(OUT))]=sha(preview)
    return result
