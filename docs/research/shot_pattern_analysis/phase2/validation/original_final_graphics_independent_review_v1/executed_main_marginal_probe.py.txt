import gzip,csv,json,hashlib
from pathlib import Path
import numpy as np
from matplotlib.backends.backend_agg import FigureCanvasAgg
from src.tools.shot_pattern_analysis import academic_plots as ap
root=Path('/home/dieterolson/.codex/worktrees/shot-pattern-analysis/UpstreamDrift/docs/research/shot_pattern_analysis/phase2')
cell=root/'final_v1/driver_lean5_curve2_rotation2_combined2'
points={n:[] for n in ('Straight','Draw','Fade')}
with gzip.open(cell/'shots.csv.gz','rt') as f:
 for r in csv.DictReader(f):
  if r['status']=='supported':points[r['pattern']].append([float(r['carry_x_m']),float(r['carry_y_m'])])
points={n:np.array(p) for n,p in points.items()}
cap=[]
def save(fig,stem,*,dpi):cap.append(fig);return {}
ap._save_formats=save
ap.write_academic_dispersion_figure(points,Path('/tmp/no-write'),scenario='Read-Only Actual Endpoint Tick Audit',metadata={},target_point_m=(json.loads((cell/'summary.json').read_text())['target_x_m'],0),presentation='academic',dpi=180)
f=cap[0];FigureCanvasAgg(f);f.canvas.draw();renderer=f.canvas.get_renderer();items=[]
for i in range(3):
 main,right=f.axes[3*i+1:3*i+3]
 a=[t for t in main.get_xticklabels() if t.get_visible() and t.get_text() and main.get_xlim()[0]<=t.get_position()[0]<=main.get_xlim()[1]]
 b=[t for t in right.get_xticklabels() if t.get_visible() and t.get_text() and right.get_xlim()[0]<=t.get_position()[0]<=right.get_xlim()[1]]
 left=max(a,key=lambda t:t.get_window_extent(renderer).x1);r=min(b,key=lambda t:t.get_window_extent(renderer).x0)
 items.append({'pattern':list(points)[i],'main_tick':left.get_text(),'marginal_tick':r.get_text(),'gap_points':(r.get_window_extent(renderer).x0-left.get_window_extent(renderer).x1)*72/f.dpi})
print(json.dumps({'renderer_sha256':hashlib.sha256(Path(ap.__file__).read_bytes()).hexdigest(),'cell':cell.name,'rows':{k:len(v) for k,v in points.items()},'measurement':items},indent=2))
