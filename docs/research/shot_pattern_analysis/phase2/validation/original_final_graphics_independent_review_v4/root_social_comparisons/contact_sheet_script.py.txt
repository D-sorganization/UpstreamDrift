from pathlib import Path
from PIL import Image, ImageOps, ImageDraw
import json, hashlib
root=Path('/home/dieterolson/.codex/worktrees/shot-pattern-analysis/UpstreamDrift/docs/research/shot_pattern_analysis/phase2')
source=root/'final_figures_v1'
out=root/'validation/original_final_graphics_independent_review_v4/root_social_comparisons'
out.mkdir(exist_ok=True)
files=sorted((source/'social/comparisons').rglob('*.png'))
assert len(files)==30
rows=[]
for f in files:
 with Image.open(f) as im:
  im.load()
  rows.append({'path':str(f.relative_to(source)), 'sha256':hashlib.sha256(f.read_bytes()).hexdigest(),'pixels':list(im.size)})
for k in range(0,len(files),6):
 canvas=Image.new('RGB',(1800,1720),'#eeeeee');draw=ImageDraw.Draw(canvas)
 for n,f in enumerate(files[k:k+6]):
  x=(n%3)*600;y=(n//3)*860
  with Image.open(f) as im:
   thumb=ImageOps.contain(im.convert('RGB'),(592,804))
   canvas.paste(thumb,(x+(600-thumb.width)//2,y+48))
  label=str(f.relative_to(source/'social/comparisons'))
  draw.text((x+5,y+5),f'{k+n+1}: {label}',fill='black')
 canvas.save(out/f'contact_{k//6+1:02}.jpg',quality=92)
(out/'inventory.json').write_text(json.dumps({'files':rows},indent=2)+'\n')
print(out)
